Esquema Mecatrónico y Código C++

Diseñado en Tinkercad Circuits con lenguaje C++ nativo para Arduino Uno R3.

Arquitectura de Conexión en Protoboard

  • Microcontrolador: Arduino Uno R3 / Nano 3.0 operando a 5V/6V regulados.
  • Sensor LDR: Entrada Analógica A0 mediante divisor de tensión.
  • Sensor PIR: Entrada Digital Pin 2 para detección infrarroja de movimiento.
  • Botón Manual: Entrada Digital Pin 3 en modo INPUT_PULLUP interno.
  • Zumbador / Buzzer: Salida Digital Pin 6 con señal de modulación acústica.
  • Luminaria RGB: Salidas PWM en Pin 9 (Rojo), Pin 10 (Verde), Pin 11 (Azul).
proyecto_ruta.cpp (C++ Tinkercad) 
// =================================================================
// PROYECTO: Sistema Eficiente de Iluminación Nocturna - IRTRA
// LENGUAJE: C++ Puro (Tinkercad Circuits)
// =================================================================

const int PIN_LDR = A0;       // Sensor de Luz Ambiental
const int PIN_PIR = 2;        // Sensor de Movimiento PIR
const int PIN_BUTTON = 3;     // Botón Mantenimiento / Emergencia

const int PIN_BUZZER = 6;     // Buzzer para señales sonoras
const int PIN_RGB_R = 9;      // Pin Rojo (PWM)
const int PIN_RGB_G = 10;     // Pin Verde (PWM)
const int PIN_RGB_B = 11;     // Pin Azul (PWM)

const int UMBRAL_NOCHE = 500; // Umbral de Oscuridad

void setup() {
  pinMode(PIN_PIR, INPUT);
  pinMode(PIN_BUTTON, INPUT_PULLUP);
  
  pinMode(PIN_BUZZER, OUTPUT);
  pinMode(PIN_RGB_R, OUTPUT);
  pinMode(PIN_RGB_G, OUTPUT);
  pinMode(PIN_RGB_B, OUTPUT);
  
  Serial.begin(9600);
}

void establecerColor(int rojo, int verde, int azul) {
  analogWrite(PIN_RGB_R, rojo);
  analogWrite(PIN_RGB_G, verde);
  analogWrite(PIN_RGB_B, azul);
}

void loop() {
  int nivelLuz = analogRead(PIN_LDR);
  int movimiento = digitalRead(PIN_PIR);
  int botonPresionado = (digitalRead(PIN_BUTTON) == LOW);

  Serial.print("Nivel de Luz: ");
  Serial.print(nivelLuz);
  Serial.print(" | Movimiento: ");
  Serial.println(movimiento);

  // MODO 1: Botón de Mantenimiento Presionado (Prioridad)
  if (botonPresionado) {
    establecerColor(0, 0, 255); // Luz Azul (Modo Mantenimiento)
    tone(PIN_BUZZER, 800, 100);  // Tono de confirmación
    delay(200);
    return;
  }

  // MODO 2: Es de Noche
  if (nivelLuz < UMBRAL_NOCHE) {
    if (movimiento == HIGH) {
      establecerColor(255, 255, 255); // Blanco 100% Presencia
      noTone(PIN_BUZZER);
    } else {
      establecerColor(0, 0, 0);       // Reposo Ahorro
      noTone(PIN_BUZZER);
    }
  } 
  // MODO 3: Es de Día
  else {
    establecerColor(0, 0, 0);         // Apagado Total
    noTone(PIN_BUZZER);
  }

  delay(100);
}
